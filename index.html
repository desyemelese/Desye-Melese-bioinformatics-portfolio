gromacs_md_production.sh
SLURM Batch Executable 
# 1. Energy Minimization & System Equilibration
gmx grompp -f em.mdp -c complex.gro -p topol.top -o em.tpr
gmx mdrun -v -deffnm em

# 2. Extended MD Simulation Production
gmx grompp -f md_200ns.mdp -c npt.gro -t npt.cpt -p topol.top -o md_200.tpr
gmx mdrun -deffnm md_200 -nb gpu -pme gpu

# 3. Binding Free Energy Calculation (MM/PBSA)
gmx_MMPBSA -O -i mmpbsa.in -cs md_200.tpr -ci index.ndx -cg 1 2 -ct md_200_noPBC.xtc
